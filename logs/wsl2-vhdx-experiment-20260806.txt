fill:    mkdir -p ~/fakecache && cd ~/fakecache && for i in 1 2 3 4; do dd if=/dev/urandom of=blob_$i bs=4M count=1280 status=none; done

compact.dpt:

  attach vdisk readonly
  compact vdisk
  detach vdisk
  exit
